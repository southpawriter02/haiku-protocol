## Evaluation Criteria Matrix

### Scoring Schema (1–5 scale)

```
┌────────────────────────────────────────┐
│ Evaluation Criteria (Weighted)         │
├────────────────────────────────────────┤
│                                        │
│ 1. Compression Ratio         (20%)     │
│    Token reduction, semantic loss      │
│                                        │
│ 2. Semantic Preservation     (20%)     │
│    Quality of output, meaning loss     │
│                                        │
│ 3. Speed / Latency           (15%)     │
│    Processing time, throughput         │
│                                        │
│ 4. Approach Type             (10%)     │
│    Extractive/Abstractive/Hybrid       │
│                                        │
│ 5. Language Support          (10%)     │
│    Multilingual capability             │
│                                        │
│ 6. LLM Dependency            (10%)     │
│    Standalone vs. proprietary model    │
│                                        │
│ 7. Open Source Status        (10%)     │
│    MIT/Apache/Proprietary/Free         │
│                                        │
│ 8. Documentation Quality     (5%)      │
│    Clarity, examples, API coverage     │
│                                        │
└────────────────────────────────────────┘
```

### Detailed Criteria Definitions

| Criterion | Score 1 | Score 3 | Score 5 | Weight |
|-----------|---------|---------|---------|--------|
| **Compression Ratio** | <30% reduction | 30–60% reduction | >70% reduction | 20% |
| **Semantic Preservation** | >20% quality loss | 5–20% loss | <5% loss | 20% |
| **Speed/Latency** | >10s per 1K tokens | 1–10s per 1K tokens | <1s per 1K tokens | 15% |
| **Approach Type** | Single-method | Multi-method | Adaptive hybrid | 10% |
| **Language Support** | English only | 2–5 languages | 10+ languages | 10% |
| **LLM Dependency** | Requires GPT-4 | Works with GPT-3.5 | Model-agnostic | 10% |
| **Open Source** | Closed/Proprietary | Partial source | Full open (MIT/Apache) | 10% |
| **Documentation** | Minimal/unclear | Good coverage | Excellent with examples | 5% |

## Python Script for Tool Capability Comparison

```python
from dataclasses import dataclass, field
from typing import Optional, Dict
import json
from enum import Enum

class ApproachType(Enum):
    EXTRACTIVE = "Extractive"
    ABSTRACTIVE = "Abstractive"
    HYBRID = "Hybrid"
    CNL_GUIDED = "CNL-Guided"

@dataclass
class ToolEvaluation:
    """Represents evaluation of a compression tool."""

    name: str
    vendor: str
    version: str
    installation_status: str  # "Not tested", "Success", "Failed"

    # Criteria scores (1-5)
    compression_ratio: int
    semantic_preservation: int
    speed_latency: int
    approach_type: ApproachType
    language_support: int
    llm_dependency: int
    open_source: int
    documentation_quality: int

    # Additional metadata
    url: str
    license: Optional[str]
    cnl_implemented: bool = False
    notes: str = ""
    tested_date: str = ""

    # Weights for calculation
    weights: Dict[str, float] = field(default_factory=lambda: {
        "compression_ratio": 0.20,
        "semantic_preservation": 0.20,
        "speed_latency": 0.15,
        "approach_type": 0.10,
        "language_support": 0.10,
        "llm_dependency": 0.10,
        "open_source": 0.10,
        "documentation_quality": 0.05,
    })

    def calculate_weighted_score(self) -> float:
        """Calculate weighted overall score (1-5)."""
        scores = {
            "compression_ratio": self.compression_ratio,
            "semantic_preservation": self.semantic_preservation,
            "speed_latency": self.speed_latency,
            "approach_type": self.approach_type.value if isinstance(self.approach_type, ApproachType) else 3,
            "language_support": self.language_support,
            "llm_dependency": self.llm_dependency,
            "open_source": self.open_source,
            "documentation_quality": self.documentation_quality,
        }

        weighted_sum = sum(scores[k] * self.weights[k] for k in scores)
        return round(weighted_sum, 2)

    def to_markdown_row(self) -> str:
        """Generate a markdown table row."""
        score = self.calculate_weighted_score()
        cnl_mark = "✅" if self.cnl_implemented else "❌"
        return (
            f"| {self.name} | {self.compression_ratio}/5 | {self.semantic_preservation}/5 | "
            f"{self.speed_latency}/5 | {self.approach_type.value} | {self.language_support}/5 | "
            f"{self.llm_dependency}/5 | {self.open_source}/5 | {self.documentation_quality}/5 | "
            f"**{score}** | {cnl_mark} |"
        )

def generate_competitive_matrix(evaluations: list[ToolEvaluation]) -> str:
    """Generate markdown-formatted competitive analysis matrix."""

    header = (
        "| Tool | Compression | Semantics | Speed | Approach | Languages | "
        "LLM Dep | Open Src | Docs | Score | CNL |\n"
        "|------|------|------|------|------|------|------|------|------|------|------|\n"
    )

    rows = [eval.to_markdown_row() for eval in sorted(
        evaluations,
        key=lambda e: e.calculate_weighted_score(),
        reverse=True
    )]

    return header + "\n".join(rows)

def save_competitive_analysis(evaluations: list[ToolEvaluation], filepath: str):
    """Save competitive analysis to markdown file."""

    content = f"""# Competitive Analysis: Prompt Compression Tools

**Generated:** {json.dumps([e.__dict__ for e in evaluations], indent=2, default=str)}

## Summary Matrix

{generate_competitive_matrix(evaluations)}

## Gap Analysis

Based on evaluation, Haiku Protocol can differentiate by:
1. Implementing CNL-guided compression (none of evaluated tools do this)
2. Supporting 10+ languages from day 1
3. Model-agnostic operation (no vendor lock-in)
4. Superior semantic preservation (<2% quality loss)

"""
    with open(filepath, 'w') as f:
        f.write(content)
    print(f"✓ Saved competitive analysis to {filepath}")

# Example usage:
if __name__ == "__main__":
    tools = [
        ToolEvaluation(
            name="LLMLingua",
            vendor="Jiang et al. / EMNLP 2023",
            version="0.1.x",
            installation_status="Success",
            compression_ratio=5,
            semantic_preservation=4,
            speed_latency=4,
            approach_type=ApproachType.HYBRID,
            language_support=3,
            llm_dependency=4,
            open_source=5,
            documentation_quality=4,
            url="https://github.com/microsoft/LLMLingua",
            license="MIT",
            cnl_implemented=False,
            notes="Most mature tool, strong compression, good semantic preservation"
        ),
    ]

    save_competitive_analysis(tools, "research/competitive_analysis.md")
```

## Verification Script

```python
#!/usr/bin/env python3
"""Verify competitive analysis completeness."""

import re
import sys
from pathlib import Path

def verify_competitive_analysis(filepath: str = "research/competitive_analysis.md") -> bool:
    """Validate competitive analysis markdown structure."""

    required_sections = [
        "Competitive Analysis",
        "Summary Matrix",
        "Gap Analysis"
    ]

    required_columns = [
        "Tool", "Compression", "Semantics", "Speed",
        "Approach", "Languages", "LLM Dep", "Open Src", "Docs", "Score"
    ]

    try:
        with open(filepath, 'r') as f:
            content = f.read()

        # Check sections
        for section in required_sections:
            if section.lower() not in content.lower():
                print(f"✗ Missing section: {section}")
                return False

        # Check for markdown table
        table_match = re.search(r'\|.*\|.*\|.*\|', content)
        if not table_match:
            print("✗ No markdown table found")
            return False

        # Count tools evaluated
        table_lines = [line for line in content.split('\n') if line.strip().startswith('|') and '---' not in line]
        tool_count = len(table_lines) - 1  # Subtract header

        if tool_count < 3:
            print(f"✗ Found {tool_count} tools, minimum 3 required")
            return False

        print(f"✓ Found {tool_count} tools in competitive matrix")
        print("✓ All required sections present")
        print("✓ Markdown table validated")
        return True

    except FileNotFoundError:
        print(f"✗ File not found: {filepath}")
        return False

if __name__ == "__main__":
    success = verify_competitive_analysis()
    sys.exit(0 if success else 1)
```
